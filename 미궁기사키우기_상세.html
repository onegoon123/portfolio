프로젝트 개요

미궁기사 키우기는 자동 전투로 스테이지를 돌파하고, 획득한 골드와 장비로 기사를 성장시키는 2D 방치형 RPG입니다. 전투 중심의 반복 플레이에 Naninovel 기반 캐릭터 스토리를 결합해, 동료 해금과 호감도 상승이 다음 에피소드로 이어지는 구조를 만들었습니다.

이 프로젝트에서 저는 기획, 프로그래밍, 데이터 설계, UI 구성, 캐릭터 일러스트 제작, 시나리오 제작 워크플로우까지 1인 개발로 담당했습니다. 특히 반복 콘텐츠가 많은 방치형 장르 특성에 맞춰 무한 스테이지 진행, 다중 타겟 전투, 랜덤 장비 생성, 자동 저장/오프라인 보상, 스토리 스크립트 자동화를 하나의 루프로 연결하는 데 집중했습니다.

게임 루프 설계

방치형 RPG의 핵심은 플레이어가 직접 조작하지 않아도 “전투 → 보상 → 성장 → 더 높은 스테이지”가 끊기지 않는 것입니다. 이를 위해 DungeonSystem이 스테이지 시작, 적 스폰, 처치 카운트, 다음 스테이지 전환을 관리하고, PlayerStats와 EquipmentSystem이 성장치를 계산하도록 역할을 나눴습니다.

private void StartStage(int stage, int totalGoalCount, int enemiesToSpawn, bool regenerateMap)
{
    currentStage = stage;
    remainingEnemiesInStage = totalGoalCount;

    StageThemeData currentTheme = null;
    if (stageThemes != null && stageThemes.Count > 0)
    {
        int themeIndex = ((stage - 1) / 10) % stageThemes.Count;
        currentTheme = stageThemes[themeIndex];
    }

    if (regenerateMap && currentTheme != null && mapManager != null)
        mapManager.GenerateMap(currentTheme);

    if (spawnCoroutine != null) StopCoroutine(spawnCoroutine);
    spawnCoroutine = StartCoroutine(SpawnStageEnemies(currentTheme, stage, enemiesToSpawn));

    OnStageStart?.Invoke(stage);
}

스테이지 테마는 10스테이지 단위로 순환되며, 적 보상은 1.08^(stage-1) 계수와 스테이지 보정값을 함께 사용합니다. 덕분에 초반에는 성장 체감이 빠르고, 후반에는 장비와 업그레이드를 병행해야 하는 구조가 됩니다.

private EnemyData CreateEnemyDataForStage(StageThemeData theme, int stage)
{
    EnemyData data = ScriptableObject.CreateInstance<EnemyData>();
    data.level = stage;

    EnemyData template = null;
    if (theme != null && theme.enemyTemplates != null && theme.enemyTemplates.Count > 0)
        template = theme.enemyTemplates[Random.Range(0, theme.enemyTemplates.Count)];

    if (template != null)
    {
        data.characterName = template.characterName;
        data.sprite = template.sprite;
        data.baseMaxHealth = template.baseMaxHealth;
        data.baseAttackPower = template.baseAttackPower;
        data.growthHealth = template.growthHealth;
        data.growthAttack = template.growthAttack;

        float goldMultiplier = Mathf.Pow(1.08f, stage - 1);
        data.goldReward = Mathf.RoundToInt(template.goldReward * goldMultiplier) + (stage * 10);

        float expMultiplier = Mathf.Pow(1.08f, stage - 1);
        data.experienceReward = Mathf.RoundToInt(template.experienceReward * expMultiplier) + (stage * 5);
    }

    return data;
}

ICombatTarget 기반 전투 구조

초기 전투 구조는 적이 플레이어만 공격하는 단순한 형태였습니다. 하지만 동료 시스템을 붙이면서 “동료도 전투 대상이 되고, 적이 가장 가까운 대상을 선택해야 한다”는 요구가 생겼습니다. 이를 해결하기 위해 플레이어와 동료가 공통으로 구현하는 ICombatTarget 인터페이스를 만들었습니다.

public interface ICombatTarget
{
    Transform GetTransform();
    bool IsAlive();
    void TakeDamage(int damage);
    int GetCurrentHealth();
    int GetMaxHealth();
    string GetName();
}

Enemy는 플레이어와 씬의 모든 CompanionController를 캐싱한 뒤, 유효한 대상 중 가장 가까운 대상을 선택합니다. Unity 오브젝트가 파괴된 뒤 참조가 남는 상황도 고려해 UnityEngine.Object null 체크, activeInHierarchy, IsAlive(), MissingReferenceException 처리를 함께 넣었습니다.

private void FindNearestTarget()
{
    UpdateTargetCache();

    ICombatTarget nearestTarget = null;
    float nearestDistance = float.MaxValue;

    foreach (ICombatTarget target in cachedTargets)
    {
        if (!IsValidTarget(target)) continue;

        float distance = Vector2.Distance(transform.position, target.GetTransform().position);
        if (distance < nearestDistance)
        {
            nearestTarget = target;
            nearestDistance = distance;
        }
    }

    currentTarget = nearestTarget;
    if (currentTarget != null)
    {
        cachedDistanceToPlayer = nearestDistance;
        isPlayerDetected = cachedDistanceToPlayer <= enemyData.detectionRange;
        isPlayerInRange = cachedDistanceToPlayer <= enemyData.attackRange;
    }
}
private bool IsValidTarget(ICombatTarget target)
{
    if (target == null || (target as UnityEngine.Object) == null) return false;

    try
    {
        Transform targetTransform = target.GetTransform();
        if (targetTransform == null) return false;
        if (!targetTransform.gameObject.activeInHierarchy) return false;
        if (!target.IsAlive()) return false;

        return true;
    }
    catch (MissingReferenceException)
    {
        return false;
    }
}

무한 스테이지와 오브젝트 풀링

방치형 게임은 적이 반복적으로 생성되고 사라지기 때문에 Instantiate/Destroy를 그대로 사용하면 GC와 프레임 저하가 쉽게 발생합니다. 적 생성은 EnemySpawner와 PoolManager로 분리하고, 사망한 적은 제거하지 않고 비활성화 후 풀로 반환되도록 구성했습니다.

private void OnEnemyDefeated(Enemy enemy)
{
    if (enemy == null) return;

    enemy.OnEnemyDefeated -= OnEnemyDefeated;
    activeEnemies.Remove(enemy);
    remainingEnemiesInStage--;

    CheckStageClear();
}

private void CheckStageClear()
{
    if (remainingEnemiesInStage <= 0)
    {
        OnStageClear?.Invoke(currentStage);
        NextStage();
    }
}

적 AI에서도 매 프레임 모든 계산을 수행하지 않도록 거리 체크와 타겟 캐시 갱신 주기를 나눴습니다. 거리 판정은 0.1초 간격, 타겟 목록 갱신은 0.2초 간격으로 처리해 자동 전투 중 불필요한 탐색 비용을 줄였습니다.

public async void PlayStory(string scriptName, string playerNameOverride = null)

한국어 조사 커스텀 명령어

플레이어 이름을 시나리오에 자연스럽게 넣기 위해 @jong 명령어를 구현했습니다. 한글 마지막 글자의 유니코드 값을 기준으로 받침 여부를 판별하고, 받침이 있을 때와 없을 때의 조사를 자동 선택합니다.

public override UniTask Execute(AsyncToken token = default)
{
    var text = Assigned(SourceText) ? SourceText.Value : string.Empty;
    var particleWithBatchim = Assigned(WithBatchim) ? WithBatchim.Value : "은";
    var particleWithoutBatchim = Assigned(WithoutBatchim) ? WithoutBatchim.Value : "는";

    var particle = HasFinalConsonant(text) ? particleWithBatchim : particleWithoutBatchim;
    var result = AppendToSource.HasValue ? $"{text}{particle}" : particle;

    if (Assigned(TargetVariable))
    {
        var variables = Engine.GetService<ICustomVariableManager>();
        variables.SetVariableValue(TargetVariable, new(result));
    }

    return UniTask.CompletedTask;
}
private static bool HasFinalConsonant(string text)
{
    if (string.IsNullOrEmpty(text)) return false;

    var lastChar = text[text.Length - 1];
    const int hangulBase = 0xAC00;
    const int hangulLast = 0xD7A3;

    if (lastChar < hangulBase || lastChar > hangulLast)
        return false;

    var code = lastChar - hangulBase;
    var jong = code % 28;
    return jong != 0;
}

단순한 문자열 치환 대신 스크립트 명령어로 처리했기 때문에, 작성자는 시나리오마다 이름의 받침을 수동으로 고려하지 않아도 됩니다. 이는 한국어 텍스트가 많은 비주얼 노벨형 콘텐츠에서 반복적인 오탈자와 어색한 문장을 줄이는 데 도움이 되었습니다.

장비 제작 시스템

EquipmentCraftingSystem은 방치형 게임의 짧은 접속 보상을 만들기 위한 시스템입니다. 최대 5포인트를 보유하고, 60초마다 1포인트가 회복됩니다. 포인트를 사용하면 무기, 투구, 갑옷, 신발 중 하나가 랜덤 생성되고, 아이템 레벨은 플레이어 레벨을 기준으로 결정됩니다.

private ItemRarity DetermineRarity()
{
    float roll = Random.Range(0f, 1f);

    if (roll <= commonChance) return ItemRarity.Common;
    else if (roll <= uncommonChance) return ItemRarity.Uncommon;
    else if (roll <= rareChance) return ItemRarity.Rare;
    else if (roll <= epicChance) return ItemRarity.Epic;
    else return ItemRarity.Legendary;
}
private void GenerateRandomStats(ItemData item, int level, float multiplier)
{
    int statCount = Random.Range(minStats, maxStats + 1);
    List<StatType> availableStats = new List<StatType>
    {
        StatType.Health,
        StatType.Attack,
        StatType.Defense,
        StatType.AttackSpeed,
        StatType.MoveSpeed,
        StatType.CritChance,
        StatType.CritDamage
    };

    for (int i = 0; i < statCount && availableStats.Count > 0; i++)
    {
        int randomIndex = Random.Range(0, availableStats.Count);
        StatType selectedStat = availableStats[randomIndex];
        availableStats.RemoveAt(randomIndex);

        float baseStat = level * statPerLevel * multiplier;
        ApplyStat(item, selectedStat, baseStat);
    }
}

    }
}

골드는 일반 정수 대신 DarkNaku.Number를 사용해 큰 수 표기에 대응했습니다. 장비와 동료 데이터는 저장용 DTO로 변환해 보관하고, 로드 시 다시 런타임 객체로 복원합니다. 저장 버전도 함께 기록해 이후 저장 구조가 바뀌어도 최소한의 호환 처리를 할 수 있도록 했습니다.

데이터 중심 밸런싱

캐릭터와 적 스탯은 ScriptableObject에 기본값과 성장 계수를 나누어 저장했습니다. HP, 공격력, 방어력 같은 장기 성장 수치는 지수 함수를 사용하고, 치명타 확률은 0~1 범위로 클램프해 비정상적인 값이 나오지 않도록 처리했습니다.

Stat(level) = baseValue * growthRate^(level - 1)

private int CalculateStat(int baseValue, float growth, int level)
{
    if (level <= 1) return baseValue;

    double result = baseValue * System.Math.Pow(growth, level - 1);
    if (result > int.MaxValue) return int.MaxValue;

    return (int)result;
}

이 구조 덕분에 코드 수정 없이 Inspector와 CSV 값을 조정해 성장 속도, 적 난이도, 보상량을 빠르게 튜닝할 수 있습니다.
